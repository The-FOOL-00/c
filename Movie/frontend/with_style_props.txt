App.vue
----------------------------
<script setup>
import { ref } from 'vue'
import Booking from './components/Booking.vue'

const bookings = ref([])
const showBookings = ref(false)

async function toggleBookings() {
  showBookings.value = !showBookings.value

  if (showBookings.value) {
    const response = await fetch('http://localhost:8080/api/bookings')
    bookings.value = await response.json()
  }
}
</script>

<template>
  <h1>Movie Booking System</h1>

  <button class="btn" @click="toggleBookings">
    {{ showBookings ? 'Hide Bookings' : 'View Bookings' }}
  </button>

  <Booking
    v-if="showBookings"
    :bookings="bookings"
  />
</template>

<style>
.btn {
  padding: 10px 18px;
  background: white;
  color: black;
  border: 1px solid black;
  border-radius: 5px;
  cursor: pointer;
  transition: 0.2s;
}

.btn:hover {
  background: royalblue;
  color: white;
  font-weight: bold;
}
</style>
